#include <Wire.h>
#include <MadgwickAHRS.h>
#include <PID_v1.h>
#include "esp_timer.h"

Madgwick filter;
#define MPU_ADDR 0x68

// --- PINOUT ---
#define L_RPWM 26
#define L_LPWM 25
#define L_REN 27
#define L_LEN 14
#define R_RPWM 33
#define R_LPWM 32
#define R_REN 12
#define R_LEN 13 
#define LEFT_A 18
#define LEFT_B 19
#define RIGHT_A 4
#define RIGHT_B 5

// --- PID & RAMPING VARIABLES ---
double leftSetpoint, leftInput, leftOutput;
double rightSetpoint, rightInput, rightOutput;

// Current "Smoothed" targets for the PID to prevent gear slam
double smoothedLeft = 0;
double smoothedRight = 0;

// TUNING PARAMETERS (Adjust for 10kg weight)
double Kp = 1.0, Ki = 15.0, Kd = 0.1; 
double maxAccel = 0.8; // Max change in ticks per 20ms loop

PID leftPID(&leftInput, &leftOutput, &smoothedLeft, 1.0, Ki, Kd, DIRECT);
PID rightPID(&rightInput, &rightOutput, &smoothedRight,2.0, 18.0, 0.1, DIRECT);

// --- STATE VARIABLES ---
volatile long left_ticks = 0;
volatile long right_ticks = 0;
long prev_left_ticks = 0;
long prev_right_ticks = 0;

float ax, ay, az, gx, gy, gz;
float roll, pitch, yaw;
String buffer = "";

// --- CALIBRATION ---
float ax_off, ay_off, az_off, gx_off, gy_off, gz_off;

// --- PROTOTYPES ---
void imuTask(void *pvParameters);
void motorTask(void *pvParameters);
void publishTask(void *pvParameters);

// --- ISRs ---
void IRAM_ATTR leftISR() { (digitalRead(LEFT_B)) ? left_ticks++ : left_ticks--; }
void IRAM_ATTR rightISR() { (digitalRead(RIGHT_B)) ? right_ticks++ : right_ticks--; }

// --- MPU UTILITIES ---
void setupMPU() {
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x6B); Wire.write(0);
  Wire.endTransmission(true);
}

void readRawMPU() {
  Wire.beginTransmission(MPU_ADDR);
  Wire.write(0x3B);
  Wire.endTransmission(false);
  Wire.requestFrom((uint8_t)MPU_ADDR, (uint8_t)14, (bool)true);
  
  int16_t AcX = Wire.read()<<8|Wire.read();
  int16_t AcY = Wire.read()<<8|Wire.read();
  int16_t AcZ = Wire.read()<<8|Wire.read();
  Wire.read(); Wire.read();
  int16_t GyX = Wire.read()<<8|Wire.read();
  int16_t GyY = Wire.read()<<8|Wire.read();
  int16_t GyZ = Wire.read()<<8|Wire.read();

  ax = AcX / 16384.0; ay = AcY / 16384.0; az = AcZ / 16384.0;
  gx = (GyX / 131.0) * DEG_TO_RAD; 
  gy = (GyY / 131.0) * DEG_TO_RAD; 
  gz = (GyZ / 131.0) * DEG_TO_RAD;
}

void calibrateMPU() {
  Serial.println("IMU CALIBRATING... DO NOT MOVE ROBOT");
  float s_ax=0, s_ay=0, s_az=0, s_gx=0, s_gy=0, s_gz=0;
  for(int i=0; i<1000; i++) {
    readRawMPU();
    s_ax+=ax; s_ay+=ay; s_az+=az;
    s_gx+=gx; s_gy+=gy; s_gz+=gz;
    delay(2);
  }
  ax_off=s_ax/1000; ay_off=s_ay/1000; az_off=(s_az/1000)-1.0;
  gx_off=s_gx/1000; gy_off=s_gy/1000; gz_off=s_gz/1000;
  Serial.println("CALIBRATION COMPLETE");
}

void driveMotor(int l_out, int r_out) {
  if (l_out >= 0) { analogWrite(L_RPWM, l_out); analogWrite(L_LPWM, 0); }
  else { analogWrite(L_RPWM, 0); analogWrite(L_LPWM, -l_out); }
  if (r_out >= 0) { analogWrite(R_RPWM, r_out); analogWrite(R_LPWM, 0); }
  else { analogWrite(R_RPWM, 0); analogWrite(R_LPWM, -r_out); }
}

// --- TASKS ---

void imuTask(void *pvParameters) {
  TickType_t xLastWakeTime = xTaskGetTickCount();
  for (;;) {
    readRawMPU();
    // Apply Calibration and Madgwick processing
    filter.updateIMU(gx-gx_off, gy-gy_off, gz-gz_off, ax-ax_off, ay-ay_off, az-az_off);
    roll = filter.getRoll(); pitch = filter.getPitch(); yaw = filter.getYaw();
    vTaskDelayUntil(&xLastWakeTime, pdMS_TO_TICKS(5)); 
  }
}

void motorTask(void *pvParameters) {
  TickType_t xLastWakeTime = xTaskGetTickCount();
  for (;;) {
    // 1. INPUT RAMPING (Slew Rate Limiter)
    // Limits the rate of change of the setpoint to avoid gear lash
    if (leftSetpoint > smoothedLeft) 
      smoothedLeft += fmin(maxAccel, leftSetpoint - smoothedLeft);
    else if (leftSetpoint < smoothedLeft) 
      smoothedLeft -= fmin(maxAccel, smoothedLeft - leftSetpoint);

    if (rightSetpoint > smoothedRight) 
      smoothedRight += fmin(maxAccel, rightSetpoint - smoothedRight);
    else if (rightSetpoint < smoothedRight) 
      smoothedRight -= fmin(maxAccel, smoothedRight - rightSetpoint);

    // 2. PID CALCULATION
    leftInput = (double)(left_ticks - prev_left_ticks);
    rightInput = (double)(right_ticks - prev_right_ticks);
    prev_left_ticks = left_ticks; prev_right_ticks = right_ticks;

    leftPID.Compute();
    rightPID.Compute();

    // 3. EXECUTION
    driveMotor((int)leftOutput, (int)rightOutput);
    vTaskDelayUntil(&xLastWakeTime, pdMS_TO_TICKS(20)); 
  }
}

void publishTask(void *pvParameters) {
  for (;;) {
    long t = esp_timer_get_time();
    // Output format: DATA, Time, L_Ticks, R_Ticks, Roll, Pitch, Yaw, Ax, Ay, Az, Gx, Gy, Gz
    Serial.printf("DATA,%ld,%ld,%ld,%.2f,%.2f,%.2f,%.4f,%.4f,%.4f,%.4f,%.4f,%.4f\n", 
                  t, left_ticks, right_ticks, roll, pitch, yaw, ax-ax_off, ay-ay_off, az-az_off, gx-gx_off, gy-gy_off, gz-gz_off);
    vTaskDelay(pdMS_TO_TICKS(20));
  }
}

void setup() {
  Serial.begin(115200);
  Wire.begin();
  setupMPU();
  calibrateMPU();
  filter.begin(200);

  leftPID.SetMode(AUTOMATIC); leftPID.SetOutputLimits(-255, 255); leftPID.SetSampleTime(20);
  rightPID.SetMode(AUTOMATIC); rightPID.SetOutputLimits(-255, 255); rightPID.SetSampleTime(20);

  int pins[] = {L_RPWM, L_LPWM, L_REN, L_LEN, R_RPWM, R_LPWM, 12, 13};
  for(int p : pins) pinMode(p, OUTPUT);
  digitalWrite(L_REN, HIGH); digitalWrite(L_LEN, HIGH);
  digitalWrite(R_REN, HIGH); digitalWrite(R_LEN, HIGH);

  pinMode(LEFT_A, INPUT_PULLUP); pinMode(LEFT_B, INPUT_PULLUP);
  pinMode(RIGHT_A, INPUT_PULLUP); pinMode(RIGHT_B, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(LEFT_A), leftISR, RISING);
  attachInterrupt(digitalPinToInterrupt(RIGHT_A), rightISR, RISING);

  // Distribute tasks across dual cores for maximum stability
  xTaskCreatePinnedToCore(imuTask, "imuTask", 4096, NULL, 3, NULL, 1);
  xTaskCreatePinnedToCore(motorTask, "motorTask", 4096, NULL, 2, NULL, 1);
  xTaskCreatePinnedToCore(publishTask, "publishTask", 4096, NULL, 1, NULL, 1);
}

void loop() {
  while (Serial.available()) {
    char c = Serial.read();
    if (c == '\n') {
      if (buffer.startsWith("CMD")) {
        int first = buffer.indexOf(','); int second = buffer.indexOf(',', first + 1);
        if (first != -1 && second != -1) {
          leftSetpoint = buffer.substring(first + 1, second).toDouble();
          rightSetpoint = buffer.substring(second + 1).toDouble();
        }
      }
      buffer = "";
    } else { buffer += c; }
  }
}
